aux_source_directory(. rime_test_src)
set(EXECUTABLE_OUTPUT_PATH ${PROJECT_BINARY_DIR}/test)
add_executable(rime_test ${rime_test_src})
target_link_libraries(rime_test
  ${rime_library}
  ${rime_dict_library}
  ${rime_gears_library}
  ${GTEST_LIBRARIES})
if(BUILD_SHARED_LIBS)
  target_compile_definitions(rime_test PRIVATE RIME_IMPORTS)
endif(BUILD_SHARED_LIBS)
# fork: opencc is vendored and built with OPENCC_BUILD_DATA=OFF (deps.mk), so no
# platform has an installed opencc dict directory to locate -- and the engine
# build graph must not grow host Python or a host-built opencc_dict to make one.
# rime_test converts against the committed mirror of opencc's own install dir
# instead, regenerated by `make -C dict/tools test-opencc` and compared
# byte-for-byte by `make -C dict/tools check`. Used on every
# platform, including the ones where a system opencc exists, so the test asserts
# the same bytes everywhere. Upstream's check below becomes an existence check on
# that directory rather than on a package finding.
set(Opencc_DICT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/opencc")
if(NOT EXISTS "${Opencc_DICT_DIR}/t2s.json")
  message(FATAL_ERROR "Could not locate the committed opencc dict directory "
                      "under test/opencc (e.g. t2s.json); required to build rime_test.")
endif()
target_compile_definitions(rime_test PRIVATE
  RIME_OPENCC_DICT_DIR="${Opencc_DICT_DIR}")

file(GLOB test_data_files ${PROJECT_SOURCE_DIR}/data/test/*.yaml)
file(COPY ${test_data_files} DESTINATION ${EXECUTABLE_OUTPUT_PATH})

add_test(NAME rime_test
  COMMAND rime_test
  WORKING_DIRECTORY ${EXECUTABLE_OUTPUT_PATH})
